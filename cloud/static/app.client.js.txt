  // ── Routing & Events ───────────────────────────
  let current = "uebersicht";
  async function render() {
    const [path, qs] = (location.hash.slice(2) || "uebersicht").split("?");
    current = pages[path] ? path : "uebersicht";
    document.querySelectorAll(".pages a").forEach((a) => a.classList.toggle("active", a.dataset.page === current));
    const params = new URLSearchParams(qs || "");
    const html = await pages[current](params);
    $("#view").innerHTML = html;
    if (params.get("p")) projectDrawer(params.get("p"));
    if (params.get("d")) diskDrawer(params.get("d"));
  }
  window.addEventListener("hashchange", () => { closeDrawer(); render(); window.scrollTo(0, 0); });
  let qt; $("#q").addEventListener("input", () => {
    clearTimeout(qt);
    qt = setTimeout(() => { if (current === "uebersicht" || current === "protokoll") location.hash = "#/projekte"; else render(); }, 180);
  });
  $("#drawer-close").addEventListener("click", closeDrawer);
  $("#scrim").addEventListener("click", closeDrawer);
  document.addEventListener("keydown", (e) => { if (e.key === "Escape") closeDrawer(); if (e.key === "/" && document.activeElement.tagName !== "INPUT") { e.preventDefault(); $("#q").focus(); } });

  document.addEventListener("click", async (e) => {
    const t = e.target.closest("[data-reset-goal],[data-age],[data-one],[data-bulk],[data-sort],[data-proj],[data-disk],[data-pf],[data-tier],[data-prio],[data-more],[data-fstate],[data-save-assign],[data-clear-assign]");
    if (!t) return;
    if (t.dataset.sort) {
      const [table, key] = t.dataset.sort.split(":");
      const st = sortState[table];
      // erster Klick: sinnvolle Richtung (Text/Datum aufsteigend bzw. Datum neueste zuerst, Zahlen größte zuerst)
      if (st.key === key) st.dir *= -1; else { st.key = key; st.dir = ["name", "tier", "holder"].includes(key) ? 1 : -1; }
      store.set("nxt-sort", sortState); return render();
    }
    if (t.dataset.proj !== undefined) { e.preventDefault(); projectDrawer(t.dataset.proj); return; }
    if (t.dataset.disk !== undefined) { e.preventDefault(); diskDrawer(t.dataset.disk); return; }
    if (t.dataset.pf) { ui.proj.filter = t.dataset.pf; ui.proj.limit = 150; history.replaceState(null, "", "#/projekte"); return render(); }
    if (t.dataset.tier) { ui.disk.tier = t.dataset.tier; return render(); }
    if (t.dataset.prio) { ui.find.prio = t.dataset.prio; ui.find.type = ""; return render(); }
    if (t.dataset.more) { if (t.dataset.more === "proj") ui.proj.limit += 300; else if (t.dataset.more === "clean") ui.clean.limit += 200; else ui.assign.limit += 200; return render(); }
    if (t.dataset.resetGoal !== undefined) {
      e.preventDefault();
      if (!confirm("Aufräum-Ziel auf den heutigen Stand setzen? Der Fortschritt startet dann bei 0 %.")) return;
      try { await post("/api/cleanup/baseline", {}); toast("Ziel neu gesetzt"); await load(); render(); } catch (err) { toast(err.message, true); }
      return;
    }
    if (t.dataset.age) { ui.clean.age = t.dataset.age; ui.clean.limit = 120; return render(); }
    if (t.dataset.one || t.dataset.bulk) {
      const keys = t.dataset.one ? [t.dataset.key] : [...ui.clean.sel];
      const level = t.dataset.one || t.dataset.bulk;
      t.disabled = true;
      try {
        const r = await post("/api/rules/bulk", { rule_keys: keys, sicherung: level });
        ui.clean.sel.clear();
        toast(`${r.count} ${r.count === 1 ? "Projekt" : "Projekte"}: ${S.meta.sicherung[level].label}`); await load(); render();
      } catch (err) { toast(err.message, true); t.disabled = false; }
      return;
    }
    if (t.dataset.fstate) {
      let note = null;
      if (t.dataset.fstate !== "offen") { note = prompt(t.dataset.fstate === "umgesetzt" ? "Was wurde gemacht? (optional)" : "Warum ignorieren? (optional)"); if (note === null) return; }
      try { await post("/api/finding", { finding_key: t.dataset.key, status: t.dataset.fstate, note }); toast("Gespeichert"); await load(); render();

      const list = S.unassigned.filter((u) => !s || [u.name, u.disk, u.soll_name].some((x) => String(x || "").toLowerCase().includes(s)));

  // ── Laufende Arbeiten: Balken oben, alle 10 s wenn etwas läuft, sonst jede Minute ──
  const KIND = { inventur: "Erfassung", scan: "Scan", hash: "Prüfsummen", kopie: "Kopie", lesetest: "Lesetest", vorgang: "Vorgang", pruefung: "Prüfung" };
  // Fortschritt nach Bytes (Kopien, Prüfungen) oder nach Anzahl (Dateien, Ordner, Schritte)
  const byBytes = (j) => !j.total_files && j.total_bytes > 0;
  const eta = (j, rate) => {
    const [done, total] = byBytes(j) ? [j.done_bytes || 0, j.total_bytes] : [j.done_files, j.total_files];
    if (!total || !rate) return "";
    const left = (total - done) / rate;
    return left > 0 ? ` · noch ~${left > 5400 ? Math.round(left / 3600) + " Std" : Math.max(1, Math.round(left / 60)) + " Min"}` : "";
  };
  async function pollJobs() {
    let data;
    try { data = await (await fetch("/api/progress")).json(); } catch { return setTimeout(pollJobs, 60000); }
    const box = $("#jobs");
    const all = [...data.jobs, ...data.stalled.map((j) => ({ ...j, _stalled: true }))];
    box.hidden = !all.length;
    box.innerHTML = all.map((j) => {
      const secs = Math.max(1, (Date.parse(j.updated_at) - Date.parse(j.started_at)) / 1000);
      const rate = (byBytes(j) ? (j.done_bytes || 0) : j.done_files) / secs;
      const pct = byBytes(j) ? Math.min(100, ((j.done_bytes || 0) / j.total_bytes) * 100)
        : j.total_files ? Math.min(100, (j.done_files / j.total_files) * 100) : null;
      const est = j.total_files && j.state === "running" && j.done_files > j.total_files;  // Schätzung überholt
      const status = j._stalled ? "keine Meldung seit " + rel(j.updated_at) : j.state === "done" ? "fertig " + rel(j.updated_at) : j.state === "failed" ? "abgebrochen" : "";
      return `<div class="job ${j.state} ${j._stalled ? "failed" : ""}"><div class="top">
        <span><b>${esc(j.disk_name || "?")}</b> · ${esc(KIND[j.kind] || j.kind)} <span class="faint">auf ${esc(j.host || "?")}</span></span>
        <span class="num muted">${byBytes(j) ? `${fmtGb((j.done_bytes || 0) / 1e9)} von ${fmtGb(j.total_bytes / 1e9)}`
          : j.kind === "vorgang" ? `Schritt ${fmtN(j.done_files)} von ${fmtN(j.total_files || 0)}`
          : `${fmtN(j.done_files)}${j.total_files ? ` von ${est || j.kind === "scan" ? "" : "~"}${fmtN(j.total_files)}` : ""} ${j.kind === "scan" ? "Ordner" : "Dateien · " + fmtGb((j.done_bytes || 0) / 1e9)}`}
          ${pct != null && j.state === "running" && !est ? ` · <b style="color:var(--text)">${Math.round(pct)} %</b>` : ""}${j.state === "running" && !j._stalled && !est ? eta(j, rate) : ""}
          ${status ? " · " + esc(status) : ""}${j.message ? " · " + esc(j.message) : ""}</span></div>
        <div class="bar ${pct == null || est ? (j.state === "running" && !j._stalled ? "indet" : "") : ""}"><span style="width:${j.state === "done" ? 100 : pct ?? 0}%"></span></div></div>`;
    }).join("");
    setTimeout(pollJobs, data.jobs.some((j) => j.state === "running") ? 10000 : 60000);
  }
  pollJobs();

  // ── Rückfragen des Assistenten ─────────────────
  async function pollQuestions() {

    const open = rows.filter((q) => q.status === "offen");

    }).join("");
    setTimeout(pollQuestions, open.length ? 15000 : 60000);
  }

    } catch {}
  }, 60000);
})();
